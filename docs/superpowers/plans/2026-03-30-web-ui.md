```go
// Copyright 2024
// SPDX-License-Identifier: Apache-2.0

package library

import (
	"context"
	"fmt"

	"github.com/penny-vault/pv-data/data"
	"github.com/rs/zerolog/log"
)

// StatusToString converts a data.StatusType to the string stored in run_history.
func StatusToString(s data.StatusType) string {
	if s == data.RunSuccess {
		return "success"
	}
	return "failed"
}

// SaveRunHistory persists a RunSummary to the run_history table.
func (myLibrary *Library) SaveRunHistory(ctx context.Context, summary data.RunSummary) error {
	conn, err := myLibrary.Pool.Acquire(ctx)
	if err != nil {
		return fmt.Errorf("acquire connection for run_history: %w", err)
	}
	defer conn.Release()

	_, err = conn.Exec(ctx,
		`INSERT INTO run_history (subscription_id, start_time, end_time, num_observations, status)
		 VALUES ($1, $2, $3, $4, $5)`,
		summary.SubscriptionID,
		summary.StartTime,
		summary.EndTime,
		summary.NumObservations,
		StatusToString(summary.Status),
	)
	if err != nil {
		return fmt.Errorf("insert run_history: %w", err)
	}

	log.Info().
		Str("subscription", summary.SubscriptionName).
		Str("status", StatusToString(summary.Status)).
		Int("observations", summary.NumObservations).
		Msg("saved run history")

	return nil
}

// RunHistoryEntry represents a single row from the run_history table.
type RunHistoryEntry struct {
	ID             string `json:"id" db:"id"`
	SubscriptionID string `json:"subscription_id" db:"subscription_id"`
	StartTime      string `json:"start_time" db:"start_time"`
	EndTime        string `json:"end_time" db:"end_time"`
	NumObservations int   `json:"num_observations" db:"num_observations"`
	Status         string `json:"status" db:"status"`
	CreatedOn      string `json:"created_on" db:"created_on"`
}

// RunHistory returns paginated run history for a subscription.
func (myLibrary *Library) RunHistory(ctx context.Context, subscriptionID string, limit, offset int) ([]RunHistoryEntry, int, error) {
	conn, err := myLibrary.Pool.Acquire(ctx)
	if err != nil {
		return nil, 0, fmt.Errorf("acquire connection: %w", err)
	}
	defer conn.Release()

	var total int
	err = conn.QueryRow(ctx,
		`SELECT count(*) FROM run_history WHERE subscription_id::text LIKE $1`,
		subscriptionID+"%",
	).Scan(&total)
	if err != nil {
		return nil, 0, fmt.Errorf("count run_history: %w", err)
	}

	rows, err := conn.Query(ctx,
		`SELECT id, subscription_id, start_time, end_time, num_observations, status, created_on
		 FROM run_history WHERE subscription_id::text LIKE $1
		 ORDER BY start_time DESC LIMIT $2 OFFSET $3`,
		subscriptionID+"%", limit, offset,
	)
	if err != nil {
		return nil, 0, fmt.Errorf("query run_history: %w", err)
	}
	defer rows.Close()

	var entries []RunHistoryEntry
	for rows.Next() {
		var e RunHistoryEntry
		if err := rows.Scan(&e.ID, &e.SubscriptionID, &e.StartTime, &e.EndTime, &e.NumObservations, &e.Status, &e.CreatedOn); err != nil {
			return nil, 0, fmt.Errorf("scan run_history row: %w", err)
		}
		entries = append(entries, e)
	}

	return entries, total, nil
}

// SparklineData represents aggregated daily observation counts for sparkline rendering.
type SparklineData struct {
	Date            string `json:"date"`
	NumObservations int    `json:"num_observations"`
}

// RunHistorySparkline returns daily aggregated observation counts for the last 30 days.
func (myLibrary *Library) RunHistorySparkline(ctx context.Context, subscriptionID string) ([]SparklineData, error) {
	conn, err := myLibrary.Pool.Acquire(ctx)
	if err != nil {
		return nil, fmt.Errorf("acquire connection: %w", err)
	}
	defer conn.Release()

	rows, err := conn.Query(ctx,
		`SELECT start_time::date AS date, sum(num_observations) AS num_observations
		 FROM run_history
		 WHERE subscription_id::text LIKE $1
		   AND start_time >= now() - interval '30 days'
		 GROUP BY start_time::date
		 ORDER BY date`,
		subscriptionID+"%",
	)
	if err != nil {
		return nil, fmt.Errorf("query sparkline: %w", err)
	}
	defer rows.Close()

	var result []SparklineData
	for rows.Next() {
		var s SparklineData
		if err := rows.Scan(&s.Date, &s.NumObservations); err != nil {
			return nil, fmt.Errorf("scan sparkline row: %w", err)
		}
		result = append(result, s)
	}

	return result, nil
}
```

```go
// Copyright 2024
// SPDX-License-Identifier: Apache-2.0

package web

import (
	"context"
	"strings"
	"time"

	"github.com/gofiber/fiber/v2"
	"github.com/golang-jwt/jwt/v5"
	"github.com/MicahParks/keyfunc/v3"
	"github.com/rs/zerolog/log"
	"github.com/spf13/viper"
)

// NewAuthMiddleware creates a Fiber middleware that validates Zitadel JWTs.
// It fetches the JWKS from the Zitadel issuer and validates the token signature,
// expiration, and issuer claim.
func NewAuthMiddleware() fiber.Handler {
	authDomain := viper.GetString("auth.domain")
	clientID := viper.GetString("auth.client_id")

	if authDomain == "" {
		log.Warn().Msg("auth.domain not configured, auth middleware disabled")
		return func(c *fiber.Ctx) error {
			return c.Next()
		}
	}

	issuer := "https://" + authDomain
	jwksURL := issuer + "/oauth/v2/keys"

	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()

	jwks, err := keyfunc.NewJWKSetJSON(ctx, keyfunc.Options{
		RefreshInterval: 1 * time.Hour,
	})
	if err != nil {
		// Fall back to fetching on each request if initial fetch fails
		log.Error().Err(err).Msg("failed to fetch JWKS, will retry per-request")
	}

	_ = jwks // Used below in closure

	// Fetch JWKS URL directly
	k, err := keyfunc.NewDefault([]string{jwksURL})
	if err != nil {
		log.Fatal().Err(err).Str("url", jwksURL).Msg("failed to create JWKS keyfunc")
	}

	return func(c *fiber.Ctx) error {
		authHeader := c.Get("Authorization")
		if authHeader == "" {
			return c.Status(fiber.StatusUnauthorized).JSON(HttpError{
				Code:    "UNAUTHORIZED",
				Message: "missing Authorization header",
			})
		}

		parts := strings.SplitN(authHeader, " ", 2)
		if len(parts) != 2 || !strings.EqualFold(parts[0], "bearer") {
			return c.Status(fiber.StatusUnauthorized).JSON(HttpError{
				Code:    "UNAUTHORIZED",
				Message: "invalid Authorization header format",
			})
		}

		tokenString := parts[1]

		token, err := jwt.Parse(tokenString, k.Keyfunc, jwt.WithIssuer(issuer), jwt.WithExpirationRequired())
		if err != nil {
			log.Debug().Err(err).Msg("JWT validation failed")
			return c.Status(fiber.StatusUnauthorized).JSON(HttpError{
				Code:    "UNAUTHORIZED",
				Message: "invalid token",
			})
		}

		if !token.Valid {
			return c.Status(fiber.StatusUnauthorized).JSON(HttpError{
				Code:    "UNAUTHORIZED",
				Message: "token not valid",
			})
		}

		// Optionally validate audience/client_id
		if clientID != "" {
			claims, ok := token.Claims.(jwt.MapClaims)
			if !ok {
				return c.Status(fiber.StatusUnauthorized).JSON(HttpError{
					Code:    "UNAUTHORIZED",
					Message: "invalid claims",
				})
			}

			// Zitadel puts the client_id in the "aud" claim
			aud, _ := claims.GetAudience()
			found := false
			for _, a := range aud {
				if a == clientID {
					found = true
					break
				}
			}
			if !found {
				return c.Status(fiber.StatusUnauthorized).JSON(HttpError{
					Code:    "UNAUTHORIZED",
					Message: "token audience mismatch",
				})
			}
		}

		return c.Next()
	}
}
```

```go
type UpdateSubscriptionRequest struct {
	Schedule      *string            `json:"schedule,omitempty"`
	Config        map[string]string  `json:"config,omitempty"`
	HealthCheckID *string            `json:"health_check_id,omitempty"`
	Active        *bool              `json:"active,omitempty"`
}

func UpdateSubscription(c *fiber.Ctx) error {
	ctx := c.UserContext()
	id := c.Params("id")

	var req UpdateSubscriptionRequest
	if err := c.BodyParser(&req); err != nil {
		return c.Status(fiber.StatusBadRequest).JSON(HttpError{
			Code:    "INVALID_BODY",
			Message: "could not parse request body",
		})
	}

	myLibrary, err := getLibrary(ctx)
	if err != nil {
		return c.Status(fiber.StatusInternalServerError).JSON(HttpError{
			Code:    "DB_ERROR",
			Message: "could not connect to database",
		})
	}
	defer myLibrary.Pool.Close()

	sub, err := myLibrary.SubscriptionFromID(ctx, id)
	if err != nil {
		return c.Status(fiber.StatusNotFound).JSON(HttpError{
			Code:    "NOT_FOUND",
			Message: "subscription not found",
		})
	}

	conn, err := myLibrary.Pool.Acquire(ctx)
	if err != nil {
		return c.Status(fiber.StatusInternalServerError).JSON(HttpError{
			Code:    "DB_ERROR",
			Message: "could not acquire connection",
		})
	}
	defer conn.Release()

	if req.Schedule != nil {
		sub.Schedule = *req.Schedule
	}
	if req.Config != nil {
		sub.Config = req.Config
	}
	if req.HealthCheckID != nil {
		sub.HealthCheckID = *req.HealthCheckID
	}

	if req.Active != nil {
		if *req.Active {
			if err := sub.Activate(ctx); err != nil {
				return c.Status(fiber.StatusInternalServerError).JSON(HttpError{
					Code:    "ACTIVATE_ERROR",
					Message: err.Error(),
				})
			}
		} else {
			if err := sub.Deactivate(ctx); err != nil {
				return c.Status(fiber.StatusInternalServerError).JSON(HttpError{
					Code:    "DEACTIVATE_ERROR",
					Message: err.Error(),
				})
			}
		}
	}
```

```go
package web

import (
	"fmt"
	"regexp"
	"strconv"
	"strings"

	"github.com/gofiber/fiber/v2"
	"github.com/rs/zerolog/log"

// safeIdentifier validates that a string is a safe SQL identifier (alphanumeric + underscores).
var safeIdentifier = regexp.MustCompile(`^[a-zA-Z_][a-zA-Z0-9_]*$`)

func GetSubscriptionData(c *fiber.Ctx) error {
	ctx := c.UserContext()
	id := c.Params("id")
	datatype := c.Params("datatype")

	limit, _ := strconv.Atoi(c.Query("limit", "100"))
	offset, _ := strconv.Atoi(c.Query("offset", "0"))
	sortCol := c.Query("sort", "")
	order := c.Query("order", "asc")
	search := c.Query("q", "")

	if limit <= 0 || limit > 500 {
		limit = 100
	}
	if offset < 0 {
		offset = 0
	}
	if order != "asc" && order != "desc" {
		order = "asc"
	}

	myLibrary, err := getLibrary(ctx)
	if err != nil {
		return c.Status(fiber.StatusInternalServerError).JSON(HttpError{
			Code:    "DB_ERROR",
			Message: "could not connect to database",
		})
	}
	defer myLibrary.Pool.Close()

	sub, err := myLibrary.SubscriptionFromID(ctx, id)
	if err != nil {
		return c.Status(fiber.StatusNotFound).JSON(HttpError{
			Code:    "NOT_FOUND",
			Message: "subscription not found",
		})
	}

	tableName, ok := sub.DataTablesMap[datatype]
	if !ok {
		return c.Status(fiber.StatusBadRequest).JSON(HttpError{
			Code:    "INVALID_DATATYPE",
			Message: fmt.Sprintf("subscription does not have data type: %s", datatype),
		})
	}

	// Validate table name is safe for SQL
	if !safeIdentifier.MatchString(tableName) {
		return c.Status(fiber.StatusInternalServerError).JSON(HttpError{
			Code:    "INVALID_TABLE",
			Message: "invalid table name",
		})
	}

	conn, err := myLibrary.Pool.Acquire(ctx)
	if err != nil {
		return c.Status(fiber.StatusInternalServerError).JSON(HttpError{
			Code:    "DB_ERROR",
			Message: "could not acquire connection",
		})
	}
	defer conn.Release()

	// Build query
	query := fmt.Sprintf("SELECT * FROM %s", tableName)
	countQuery := fmt.Sprintf("SELECT count(*) FROM %s", tableName)
	var args []interface{}
	argIdx := 1

	// Text search across all text columns (simple CAST to text approach)
	if search != "" {
		query += fmt.Sprintf(" WHERE %s::text ILIKE $%d", tableName, argIdx)
		countQuery += fmt.Sprintf(" WHERE %s::text ILIKE $%d", tableName, argIdx)

		// Use a row-level text search: cast entire row to text
		query = fmt.Sprintf("SELECT * FROM %s WHERE %s::text ILIKE $%d", tableName, tableName, argIdx)
		countQuery = fmt.Sprintf("SELECT count(*) FROM %s WHERE %s::text ILIKE $%d", tableName, tableName, argIdx)
		args = append(args, "%"+search+"%")
		argIdx++
	}

	// Sort
	if sortCol != "" && safeIdentifier.MatchString(sortCol) {
		query += fmt.Sprintf(" ORDER BY %s %s", sortCol, strings.ToUpper(order))
	}

	// Pagination
	query += fmt.Sprintf(" LIMIT $%d OFFSET $%d", argIdx, argIdx+1)
	args = append(args, limit, offset)

	// Get total count
	var total int
	countArgs := args[:len(args)-2] // exclude limit/offset
	err = conn.QueryRow(ctx, countQuery, countArgs...).Scan(&total)
	if err != nil {
		log.Error().Err(err).Msg("count query failed")
		return c.Status(fiber.StatusInternalServerError).JSON(HttpError{
			Code:    "QUERY_ERROR",
			Message: "failed to count rows",
		})
	}

	// Execute data query
	rows, err := conn.Query(ctx, query, args...)
	if err != nil {
		log.Error().Err(err).Msg("data query failed")
		return c.Status(fiber.StatusInternalServerError).JSON(HttpError{
			Code:    "QUERY_ERROR",
			Message: "failed to query data",
		})
	}
	defer rows.Close()

	// Build dynamic result from column descriptions
	columns := rows.FieldDescriptions()
	colNames := make([]string, len(columns))
	for i, col := range columns {
		colNames[i] = string(col.Name)
	}

	var results []map[string]interface{}
	for rows.Next() {
		values, err := rows.Values()
		if err != nil {
			return c.Status(fiber.StatusInternalServerError).JSON(HttpError{
				Code:    "SCAN_ERROR",
				Message: "failed to scan row",
			})
		}

		row := make(map[string]interface{}, len(colNames))
		for i, name := range colNames {
			row[name] = values[i]
		}
		results = append(results, row)
	}

	return c.JSON(fiber.Map{
		"columns": colNames,
		"data":    results,
		"total":   total,
		"limit":   limit,
		"offset":  offset,
	})
}
```

```go
package web

import (

type SQLRequest struct {
	Query string `json:"query"`
}

func ExecuteSQL(c *fiber.Ctx) error {
	ctx := c.UserContext()

	var req SQLRequest
	if err := c.BodyParser(&req); err != nil {
		return c.Status(fiber.StatusBadRequest).JSON(HttpError{
			Code:    "INVALID_BODY",
			Message: "could not parse request body",
		})
	}

	if req.Query == "" {
		return c.Status(fiber.StatusBadRequest).JSON(HttpError{
			Code:    "EMPTY_QUERY",
			Message: "query cannot be empty",
		})
	}

	myLibrary, err := getLibrary(ctx)
	if err != nil {
		return c.Status(fiber.StatusInternalServerError).JSON(HttpError{
			Code:    "DB_ERROR",
			Message: "could not connect to database",
		})
	}

	columns, results, err := executeReadOnlyQuery(ctx, myLibrary, req.Query)
	if err != nil {
		log.Error().Err(err).Str("query", req.Query).Msg("SQL execution failed")
		return c.Status(fiber.StatusBadRequest).JSON(HttpError{
			Code:    "QUERY_ERROR",
			Message: err.Error(),
		})
	}

	return c.JSON(fiber.Map{
		"columns": columns,
		"data":    results,
		"count":   len(results),
	})
}

func ExportSQL(c *fiber.Ctx) error {
	ctx := c.UserContext()
	format := c.Query("format", "csv")

	var req SQLRequest
	if err := c.BodyParser(&req); err != nil {
		return c.Status(fiber.StatusBadRequest).JSON(HttpError{
			Code:    "INVALID_BODY",
			Message: "could not parse request body",
		})
	}

	if req.Query == "" {
		return c.Status(fiber.StatusBadRequest).JSON(HttpError{
			Code:    "EMPTY_QUERY",
			Message: "query cannot be empty",
		})
	}

	if err != nil {

		})
	}

	default:

	}

// executeReadOnlyQuery runs a SQL query inside a read-only transaction with a 30-second timeout.
func executeReadOnlyQuery(ctx context.Context, myLibrary *library.Library, query string) ([]string, []map[string]interface{}, error) {
	queryCtx, cancel := context.WithTimeout(ctx, 30*time.Second)
	defer cancel()

	tx, err := myLibrary.Pool.BeginTx(queryCtx, pgx.TxOptions{
		AccessMode: pgx.ReadOnly,
	})
	if err != nil {
		return nil, nil, fmt.Errorf("begin read-only transaction: %w", err)
	}
	defer tx.Rollback(queryCtx)

	rows, err := tx.Query(queryCtx, query)
	if err != nil {
		return nil, nil, fmt.Errorf("execute query: %w", err)
	}
	defer rows.Close()

	fieldDescs := rows.FieldDescriptions()
	columns := make([]string, len(fieldDescs))
	for i, fd := range fieldDescs {
		columns[i] = string(fd.Name)
	}

	var results []map[string]interface{}
	for rows.Next() {
		values, err := rows.Values()
		if err != nil {
			return nil, nil, fmt.Errorf("scan row: %w", err)
		}

		row := make(map[string]interface{}, len(columns))
		for i, name := range columns {
			row[name] = values[i]
		}
		results = append(results, row)
	}

	return columns, results, tx.Commit(queryCtx)
}

func exportCSV(c *fiber.Ctx, columns []string, results []map[string]interface{}) error {
	var buf bytes.Buffer
	w := csv.NewWriter(&buf)

	// Header
	w.Write(columns)

	// Rows
	for _, row := range results {
		record := make([]string, len(columns))
		for i, col := range columns {
			record[i] = fmt.Sprintf("%v", row[col])
		}
		w.Write(record)
	}
	w.Flush()

	c.Set("Content-Type", "text/csv")
	c.Set("Content-Disposition", "attachment; filename=query_results.csv")
	return c.Send(buf.Bytes())
}

func exportParquet(c *fiber.Ctx, columns []string, results []map[string]interface{}) error {
	// Build parquet schema dynamically: all columns as strings for simplicity.
	// This avoids type-mapping complexity and handles all PostgreSQL types.
	type Row struct{}

	// Use the low-level parquet writer with a dynamic schema
	fields := make([]parquet.Field, len(columns))
	for i, col := range columns {
		fields[i] = parquet.String(col)
	}
	schema := parquet.NewSchema("query_results", fields)

	var buf bytes.Buffer
	pw := parquet.NewGenericWriter[map[string]string](&buf, schema)

	for _, row := range results {
		stringRow := make(map[string]string, len(columns))
		for _, col := range columns {
			stringRow[col] = fmt.Sprintf("%v", row[col])
		}
		pw.Write([]map[string]string{stringRow})
	}

	if err := pw.Close(); err != nil {
		log.Error().Err(err).Msg("failed to write parquet")
		return c.Status(fiber.StatusInternalServerError).JSON(HttpError{
			Code:    "EXPORT_ERROR",
			Message: "failed to generate parquet file",
		})
	}

	c.Set("Content-Type", "application/octet-stream")
	c.Set("Content-Disposition", "attachment; filename=query_results.parquet")
	return c.Send(buf.Bytes())
}
```

**Note:** The parquet export uses a dynamic string-typed schema. This is a pragmatic choice -- PostgreSQL types vary widely and casting everything to string avoids complex type mapping while still producing valid Parquet files. A future enhancement could map PostgreSQL OIDs to Parquet types for better fidelity.

```go
//go:embed ui/dist/*
var embeddedUI embed.FS

// SetupSPA configures Fiber to serve the embedded Vue SPA with HTML5 history mode fallback.
func SetupSPA(app *fiber.App) {
	distFS, err := fs.Sub(embeddedUI, "ui/dist")
	if err != nil {
		// If dist doesn't exist (dev mode), skip static serving
		return
	}

	// Serve static assets (JS, CSS, images)
	app.Use("/assets", filesystem.New(filesystem.Config{
		Root:       http.FS(distFS),
		PathPrefix: "assets",
	}))

	// SPA fallback: serve index.html for all non-API, non-asset routes
	app.Use(func(c *fiber.Ctx) error {
		// Skip API routes
		if len(c.Path()) >= 4 && c.Path()[:4] == "/api" {
			return c.Next()
		}

		// Try to serve the file from dist
		file, err := distFS.Open(c.Path()[1:]) // strip leading /
		if err == nil {
			file.Close()
			return filesystem.New(filesystem.Config{
				Root: http.FS(distFS),
			})(c)
		}

		// Fallback to index.html for SPA routing
		data, err := fs.ReadFile(distFS, "index.html")
		if err != nil {
			return c.SendStatus(fiber.StatusNotFound)
		}

		c.Set("Content-Type", "text/html")
		return c.Send(data)
	})
}
```

- [ ] **Step 3: Create API client**

Create `web/ui/src/lib/api.ts`:

```ts
import { getAccessToken } from './oidc'

const BASE_URL = '/api/v1'

async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
  const token = await getAccessToken()

  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
    ...(options.headers as Record<string, string>),
  }

  if (token) {
    headers['Authorization'] = `Bearer ${token}`
  }

  const response = await fetch(`${BASE_URL}${path}`, {
    ...options,
    headers,
  })

  if (!response.ok) {
    const error = await response.json().catch(() => ({ message: response.statusText }))
    throw new Error(error.message || `HTTP ${response.status}`)
  }

  if (response.status === 204) {
    return undefined as T
  }

  return response.json()
}

export const api = {
  // Subscriptions
  getSubscriptions: () => request<any[]>('/subscriptions'),
  getSubscription: (id: string) => request<any>(`/subscriptions/${id}`),
  createSubscription: (data: any) => request<any>('/subscriptions', { method: 'POST', body: JSON.stringify(data) }),
  updateSubscription: (id: string, data: any) => request<any>(`/subscriptions/${id}`, { method: 'PUT', body: JSON.stringify(data) }),
  deleteSubscription: (id: string) => request<void>(`/subscriptions/${id}`, { method: 'DELETE' }),
  activateSubscription: (id: string) => request<any>(`/subscriptions/${id}/activate`, { method: 'POST' }),
  deactivateSubscription: (id: string) => request<any>(`/subscriptions/${id}/deactivate`, { method: 'POST' }),

  // Providers
  getProviders: () => request<any>('/providers'),

  // Run history
  getRunHistory: (id: string, limit = 50, offset = 0) =>
    request<any>(`/subscriptions/${id}/runs?limit=${limit}&offset=${offset}`),
  getSparkline: (id: string) => request<any[]>(`/subscriptions/${id}/runs/sparkline`),

  // Data browser
  getData: (id: string, datatype: string, params: Record<string, string> = {}) => {
    const qs = new URLSearchParams(params).toString()
    return request<any>(`/subscriptions/${id}/data/${datatype}${qs ? '?' + qs : ''}`)
  },

  // SQL
  executeSQL: (query: string) => request<any>('/sql', { method: 'POST', body: JSON.stringify({ query }) }),

  exportSQL: async (query: string, format: 'csv' | 'parquet') => {
    const token = await getAccessToken()
    const headers: Record<string, string> = { 'Content-Type': 'application/json' }
    if (token) headers['Authorization'] = `Bearer ${token}`

    const response = await fetch(`${BASE_URL}/sql/export?format=${format}`, {
      method: 'POST',
      headers,
      body: JSON.stringify({ query }),
    })

    if (!response.ok) throw new Error(`Export failed: ${response.statusText}`)

    const blob = await response.blob()
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = `query_results.${format}`
    a.click()
    URL.revokeObjectURL(url)
  },
}
```

```vue
<template>
  <div class="sparkline-container">
    <svg :width="width" :height="height" v-if="data.length > 0">
      <polyline
        :points="points"
        fill="none"
        stroke="#4589ff"
        stroke-width="1.5"

<script setup lang="ts">
import { computed } from 'vue'

const props = withDefaults(defineProps<{
  data: { date: string; num_observations: number }[]
  width?: number
  height?: number
}>(), {
  width: 120,
  height: 24,
})

const points = computed(() => {
  if (props.data.length === 0) return ''

  const values = props.data.map(d => d.num_observations)
  const max = Math.max(...values, 1)
  const min = Math.min(...values, 0)
  const range = max - min || 1

  const stepX = props.width / Math.max(values.length - 1, 1)
  const padding = 2

  return values
    .map((v, i) => {
      const x = i * stepX
      const y = padding + (props.height - 2 * padding) * (1 - (v - min) / range)
      return `${x},${y}`
    })
    .join(' ')
})
</script>
```

```vue
          <tr v-for="(row, idx) in rows" :key="idx">

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { api } from '../lib/api'

const props = defineProps<{
  subscriptionId: string
  datatype: string
  tableName: string
}>()

const columns = ref<string[]>([])
const rows = ref<any[]>([])
const total = ref(0)
const loading = ref(true)
const loadingMore = ref(false)
const search = ref('')
const sortCol = ref('')
const sortOrder = ref('asc')
const offset = ref(0)
const limit = 100
const scrollContainer = ref<HTMLElement | null>(null)

function toggleSort(col: string) {
  if (sortCol.value === col) {
    sortOrder.value = sortOrder.value === 'asc' ? 'desc' : 'asc'
  } else {
    sortCol.value = col
    sortOrder.value = 'asc'
  }
  resetAndLoad()
}

function resetAndLoad() {
  offset.value = 0
  rows.value = []
  loadData()
}

async function loadData() {
  if (loadingMore.value) return
  loadingMore.value = true

  try {
    const params: Record<string, string> = {
      limit: String(limit),
      offset: String(offset.value),
    }
    if (search.value) params.q = search.value
    if (sortCol.value) {
      params.sort = sortCol.value
      params.order = sortOrder.value
    }

    const result = await api.getData(props.subscriptionId, props.datatype, params)
    columns.value = result.columns || []
    if (offset.value === 0) {
      rows.value = result.data || []
    } else {
      rows.value.push(...(result.data || []))
    }
    total.value = result.total || 0
  } catch (e) {
    console.error('Failed to load data:', e)
  } finally {
    loading.value = false
    loadingMore.value = false
  }
}

function onScroll() {
  const el = scrollContainer.value
  if (!el) return

  // Load more when scrolled near the bottom
  const nearBottom = el.scrollHeight - el.scrollTop - el.clientHeight < 200
  if (nearBottom && rows.value.length < total.value && !loadingMore.value) {
    offset.value += limit
    loadData()
  }
}

onMounted(() => {
  loadData()
})
```
